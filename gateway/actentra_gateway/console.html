Quickstart Integration Snippet

from actentra import Boundary, guard
from actentra_core.policy import load_policy_file
from actentra_core.receipts import ReceiptStore

boundary = Boundary.from_files(
    policy_path="policies/examples/starter.yaml",
    receipts_db="receipts.db",
    session_id="agent-session-1"
)

@guard(boundary, tool="fs.read", verb="read")
def read_workspace_note(path: str) -> str:
    with open(path, "r", encoding="utf-8") as fh:
        return fh.read()
